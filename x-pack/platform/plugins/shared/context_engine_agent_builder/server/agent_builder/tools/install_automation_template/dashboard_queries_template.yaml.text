version: "1"
# Dashboard queries — one `dashboard` KI per dashboard and one `dashboard_panel` KI per
# visualization panel, each carrying the ES|QL that returns the numbers the panel shows. It is the
# same knowledge the crawler writes into the `elastic` AI index, scoped to the dashboards that read
# this AI index's data.
#
# This is a working workflow, not a sketch. Edit the `consts` block and it runs as it stands;
# everything below `consts` is already correct and does not need to be rewritten to make it work.
#
# Installed by platform.context_engine.install_automation_template. The tool fills the consts.
# Do not save this YAML yourself.
#
# There is no model call: the dashboards plugin turns each panel's configuration into ES|QL (time
# range, metrics, filters, breakdowns) and checks it runs. The verifiers on create_ki check it
# again before writing, so a KI never carries a query that does not run.
name: Dashboard queries KI automation
description: >
  Writes one Knowledge Indicator per dashboard, and per visualization panel, that reads the
  source index pattern, each carrying verified ES|QL that returns the panel's numbers.
enabled: true

consts:
  # The Context Engine AI index the indicators are written into.
  ai_index_id: __AI_INDEX_ID__
  # The human-readable name of this automation instance. Prefixes every KI ID.
  automation_name: __AUTOMATION_NAME__
  # Only panels that read this index pattern are kept (`*` globs and comma lists match).
  index_pattern: __INDEX_PATTERN__
  # The dashboards to read. Empty reads every dashboard in the space.
  dashboard_ids: __DASHBOARD_IDS__

tags:
  - dashboard-queries
  - ce-template:dashboard_queries

triggers:
  - type: manual
  # Dashboards change: re-reading them keeps the KIs in step, as the `elastic` crawler does.
  - type: scheduled
    with:
      every: 30m

steps:
  # Runs as the workflow's user, so only dashboards and data that user can read are written.
  # `use_server_info` calls this Kibana through its own listener rather than its public URL,
  # which a deployment may not be able to reach from itself.
  - name: dashboard_kis
    type: kibana.request
    with:
      use_server_info: true
      method: POST
      path: /internal/agent_builder_dashboards/ai_index_kis
      headers:
        x-elastic-internal-origin: kibana
        elastic-api-version: "1"
      body:
        index_pattern: "{{ consts.index_pattern }}"
        dashboard_ids: "${{ consts.dashboard_ids }}"

  # One KI whose write fails must not stop the others: `settled` runs every branch to a terminal
  # state. The list is capped at 100, the engine's fan-out limit.
  - name: write_kis
    type: parallel
    foreach: "{{ steps.dashboard_kis.output.kis | slice: 0, 100 | json }}"
    concurrency: 5
    mode: settled
    steps:
      # `ki_id` is stable per dashboard and panel, so a re-run replaces each KI.
      - name: create_ki
        type: context-engine.createKi
        with:
          ai_index_id: "{{ consts.ai_index_id }}"
          ki_id: "{{ consts.automation_name }}/{{ foreach.item.ki_id }}"
          ki: "${{ foreach.item.ki }}"
          verifiers:
            - esql-valid-syntax
            - esql-valid-runtime
